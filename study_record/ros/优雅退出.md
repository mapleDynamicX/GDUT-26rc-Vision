# 优雅退出

你想知道在 ROS1 的 C++ 节点中，如何捕获 Ctrl+C（SIGINT 信号）实现**优雅退出**（而非强制终止），并最终正常执行`return 0`，对吧？

```cpp
#include <ros/ros.h>
#include <signal.h>   // 信号处理头文件
#include <atomic>     // 原子变量（保证线程安全）

// 全局原子退出标志（原子变量避免多线程竞争）
std::atomic<bool> g_quit(false);

/**
 * @brief 自定义SIGINT信号处理函数（捕获Ctrl+C）
 * @param sig 接收到的信号值（SIGINT对应值为2）
 */
void sigintHandler(int sig)
{
    // 1. 打印退出提示（可选）
    ROS_INFO("收到Ctrl+C信号，开始优雅退出...");

    // 2. 执行退出前的清理逻辑（根据你的业务自定义）
    // 示例：关闭文件、释放硬件句柄、保存配置、停止发布器等
    // file.close();
    // serial_port.close();
    // publisher.shutdown();

    // 3. 设置退出标志，通知主循环退出
    g_quit = true;

    // 4. 关闭ROS节点（必须调用，否则ROS资源不会释放）
    ros::shutdown();

    // 注意：这里不要直接exit(0)，交给main函数执行return 0更优雅
}

int main(int argc, char** argv)
{
    // 1. 初始化ROS节点
    ros::init(argc, argv, "graceful_exit_node", ros::init_options::NoSigintHandler);
    // 关键：NoSigintHandler → 禁用ROS默认的SIGINT处理，改用自定义逻辑

    // 2. 创建节点句柄
    ros::NodeHandle nh;

    // 3. 注册自定义SIGINT信号处理函数（替换默认处理）
    signal(SIGINT, sigintHandler);

    // 4. 示例：创建发布器/订阅器（模拟业务逻辑）
    ros::Publisher pub = nh.advertise<std_msgs::String>("chatter", 10);
    ros::Rate rate(10);  // 10Hz循环

    // 5. 主循环（业务逻辑）
    ROS_INFO("节点启动成功，按Ctrl+C退出...");
    while (ros::ok() && !g_quit)
    {
        // 模拟业务：发布消息
        std_msgs::String msg;
        msg.data = "Hello ROS1!";
        pub.publish(msg);

        ros::spinOnce();  // 处理回调（如有订阅器则必须）
        rate.sleep();     // 按频率休眠
    }

    // 6. 退出前的最终清理（可选）
    ROS_INFO("清理完成，程序正常退出");

    // 7. 正常return 0
    return 0;
}

```
