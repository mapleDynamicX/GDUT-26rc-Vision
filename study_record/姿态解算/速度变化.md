```cpp
// 必须包含的核心头文件
#include <Eigen/Core>
// 可选：用于控制台输出
#include <iostream>

// 简化命名空间（Eigen代码必备）
using namespace Eigen;
using namespace std;

int main()
{
    // 1. 定义两个三维向量
    Vector3d v1(1, 0, 0);  // x轴单位向量
    Vector3d v2(0, 1, 0);  // y轴单位向量

    // 2. 方式1：成员函数 cross() 叉乘（推荐）
    Vector3d cross_res1 = v1.cross(v2);
    cout << "v1.cross(v2) = " << cross_res1.transpose() << endl;

    // 3. 方式2：静态函数 cross() 叉乘
    Vector3d cross_res2 = Vector3d::cross(v1, v2);
    cout << "Vector3d::cross(v1, v2) = " << cross_res2.transpose() << endl;

    return 0;
}
```
