## launch文件解释

```xml
<?xml version="1.0"?>
<!-- XML文件声明，指定遵循XML 1.0版本规范 -->
<launch>
<!-- ROS launch文件的根标签，所有配置需包含在该标签内 -->
  <!-- 定义名为"world"的参数，默认值为指定功能包下的Gazebo世界文件路径，用于指定仿真环境地图 -->
  <arg name="world" default="$(find livox_laser_simulation)/worlds/standardrobots_factory.world" />
  <!-- 定义名为"livox_sensor"的参数，默认值为指定功能包下的LiDAR传感器XACRO模型文件路径，用于指定传感器模型 -->
  <arg name="livox_sensor" default="$(find livox_laser_simulation)/urdf/livox_avia.xacro" />

  <!-- 包含gazebo_ros功能包的empty_world.launch文件，用于启动Gazebo仿真环境 -->
  <include file="$(find gazebo_ros)/launch/empty_world.launch">
    <!-- 向被包含的launch文件传递参数：指定Gazebo加载的世界文件为上面定义的"world"参数 -->
	<arg name="world_name" value="$(arg world)"/> 
    <!-- 向被包含的launch文件传递参数：仿真启动后不暂停（立即运行） -->
    <arg name="paused" value="false"/> 
    <!-- 向被包含的launch文件传递参数：ROS节点使用Gazebo的仿真时间 -->
    <arg name="use_sim_time" value="true"/> 
    <!-- 向被包含的launch文件传递参数：显示Gazebo图形界面（GUI） -->
    <arg name="gui" value="true"/> 
    <!-- 向被包含的launch文件传递参数：不启用无头模式（确保显示GUI） -->
    <arg name="headless" value="false"/> 
    <!-- 向被包含的launch文件传递参数：不启动Gazebo调试模式 -->
    <arg name="debug" value="false"/> 
    <!-- 向被包含的launch文件传递参数：Gazebo输出详细日志信息 -->
    <arg name="verbose" value="true"/>
  </include>
  
  <!-- 在ROS参数服务器创建"sensor_description"参数，通过xacro工具解析LiDAR的XACRO模型为URDF格式并存储 -->
  <param name="sensor_description" command="$(find xacro)/xacro --inorder $(arg livox_sensor)" />
  <!-- 启动gazebo_ros功能包的spawn_model节点，将参数服务器中的LiDAR URDF模型加载到Gazebo，命名为"livox_lidar" -->
  <node name="spawn_urdf" pkg="gazebo_ros" type="spawn_model" args="-urdf -model livox_lidar -param sensor_description" />
  <!-- 启动rviz功能包的rviz节点，加载指定功能包下的预设配置文件，用于可视化LiDAR仿真数据 -->
  <node name="rviz" pkg="rviz" type="rviz" args="-d $(find livox_laser_simulation)/rviz/livox_simulation.rviz"/>
</launch>

```
