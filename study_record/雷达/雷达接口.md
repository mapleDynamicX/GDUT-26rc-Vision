# 雷达接口

```bash
#!/bin/bash
# 声明脚本解释器为bash，确保脚本以bash执行而非其他shell

# 定义ROS1版本常量（只读，避免后续误修改）
readonly VERSION_ROS1="ROS1"

# 保存当前执行脚本的目录到目录栈，并重定向输出避免冗余打印
pushd `pwd` > /dev/null
# 切换到脚本所在目录（$0是脚本自身路径，dirname提取目录部分），确保路径统一
cd `dirname $0`
# 打印当前工作目录，方便确认脚本执行路径
echo "Working Path: "`pwd`

# 初始化ROS版本变量（仅用于ROS1）
ROS_VERSION=""

# 判断传入参数是否为ROS1（仅保留ROS1分支，剔除ROS2/humble逻辑）
if [ "$1" = "ROS1" ]; then
    # 确认ROS版本为ROS1
    ROS_VERSION=${VERSION_ROS1}
else
    # 非ROS1参数直接提示错误并退出
    echo "Invalid Argument: Only 'ROS1' is supported"
    exit
fi
# 打印最终确认的ROS版本，方便验证
echo "ROS version is: "$ROS_VERSION

# 清理ROS1编译生成的build目录（-rf：递归强制删除，忽略不存在的文件）
rm -rf ../../build/
# 清理ROS1编译生成的devel目录（catkin_make产物目录）
rm -rf ../../devel/
# 清理ROS2编译的install目录（冗余清理，确保无残留，不影响ROS1）
rm -rf ../../install/
# 检查上级目录的CMakeLists.txt是否存在，存在则删除（避免旧配置影响编译）
if [ -f ../CMakeLists.txt ]; then
    rm -f ../CMakeLists.txt
fi

# ===================== ROS1核心配置替换 =====================
# 检查当前目录是否有package.xml，存在则删除（避免旧配置覆盖）
if [ -f package.xml ]; then
    rm package.xml
fi
# 复制ROS1专用的package_ROS1.xml为标准package.xml（ROS编译核心配置文件）
cp -f package_ROS1.xml package.xml

# ===================== ROS1编译 =====================
# 保存当前目录到目录栈，编译后恢复
pushd `pwd` > /dev/null
# 切换到ROS工作空间根目录（脚本上级上级目录）
cd ../../
# 执行ROS1编译命令，向CMake传递ROS版本参数
catkin_make -DROS_EDITION=${VERSION_ROS1}
# 恢复到脚本所在目录（抵消上面的pushd）
popd > /dev/null

# 恢复到执行脚本前的原始目录（抵消最开始的pushd）
popd > /dev/null
```

CMakeLists.txt

```
# 声明版权信息，归属livoxtech公司（2019年）
# Copyright(c) 2019 livoxtech limited.

# 设置CMake的最低版本要求为3.0，低于该版本则CMake报错退出
# 适配ROS1生态的CMake版本兼容要求
cmake_minimum_required(VERSION 3.
0)


#---------------------------------------------------------------------------------------
# Start livox_ros_driver2 project
#---------------------------------------------------------------------------------------
# 引入cmake/version.cmake文件，该文件通常定义了版本号相关变量（如LIVOX_ROS_DRIVER2_VERSION）
include(cmake/version.cmake)

# 定义项目名称为livox_ros_driver2，指定版本号（从version.cmake读取），编译语言为C++
# VERSION：设置项目版本，LANGUAGES：指定编译语言仅为CXX（C++）
project(livox_ros_driver2 VERSION ${LIVOX_ROS_DRIVER2_VERSION} LANGUAGES CXX)

# 在CMake编译过程中输出状态信息，打印当前livox_ros_driver2的版本号
# STATUS级别会在终端显示，方便调试和确认版本
message(STATUS "livox_ros_driver2 version: ${LIVOX_ROS_DRIVER2_VERSION}")

#---------------------------------------------------------------------------------------
# Add ROS Version MACRO
#---------------------------------------------------------------------------------------
# 添加预编译宏定义BUILDING_ROS1，代码中可通过#ifdef BUILDING_ROS1区分ROS1编译逻辑
# 例如代码中#ifdef BUILDING_ROS1 { ROS1专属逻辑 }
add_definitions(-DBUILDING_ROS1)

#---------------------------------------------------------------------------------------
# find package and the dependecy
#---------------------------------------------------------------------------------------
# 查找Boost库，要求最低版本1.54，且必须找到system、thread、chrono三个组件
# REQUIRED：找不到则CMake报错退出；COMPONENTS：指定需要的Boost子组件
find_package(Boost 1.54 REQUIRED COMPONENTS
  system
  thread
  chrono
)

## Find catkin macros and libraries
## if COMPONENTS list like find_package(catkin REQUIRED COMPONENTS xyz)
## is used, also find other catkin packages
# 查找ROS1核心构建工具catkin，同时依赖指定的catkin组件
# roscpp：ROS1 C++客户端库；rospy：ROS1 Python客户端库；sensor_msgs/std_msgs：标准消息库
# message_generation：ROS1消息生成工具；rosbag：ROS1录包/回放库；pcl_ros：PCL与ROS1桥接库
find_package(catkin REQUIRED COMPONENTS
  roscpp
  rospy
  sensor_msgs
  std_msgs
  message_generation
  rosbag
  pcl_ros
)

## Find pcl lib
# 查找PCL（点云库），REQUIRED表示找不到则报错，是激光雷达点云处理的核心库
find_package(PCL REQUIRED)

## Generate messages in the 'msg' folder
# 声明需要生成的自定义消息文件，从msg目录下读取CustomPoint.msg和CustomMsg.msg
# 注释掉的Message2.msg表示预留的消息文件，暂不启用
add_message_files(FILES
  CustomPoint.msg
  CustomMsg.msg
#   Message2.msg
)

## Generate added messages and services with any dependencies listed here
# 根据add_message_files声明的消息文件，生成对应的C++/Python消息代码
# DEPENDENCIES：指定消息依赖的标准消息库（std_msgs），确保生成消息时依赖正确
generate_messages(DEPENDENCIES
  std_msgs
)

# 查找PkgConfig工具，用于查找系统级库（如APR）
find_package(PkgConfig)

# 通过PkgConfig查找apr-1库（Apache Portable Runtime，跨平台运行时库）
# pkg_check_modules是PkgConfig的CMake封装，APR是自定义变量名，代表apr-1库
pkg_check_modules(APR apr-1)

# 如果找到APR库，打印其头文件目录和库文件路径，用于调试确认
if (APR_FOUND)
  message(${APR_INCLUDE_DIRS})
  message(${APR_LIBRARIES})
endif (APR_FOUND)

###################################
## catkin specific configuration ##
###################################
## The catkin_package macro generates cmake config files for your package
## Declare things to be passed to dependent projects
## INCLUDE_DIRS: uncomment this if your package contains header files
## LIBRARIES: libraries you create in this project that dependent projects also need
## CATKIN_DEPENDS: catkin_packages dependent projects also need
## DEPENDS: system dependencies of this project that dependent projects also need
# ROS1核心配置：生成catkin包的配置文件（如XXXConfig.cmake），供依赖该包的其他ROS包使用
# CATKIN_DEPENDS：声明该包依赖的catkin包，传递给下游包；message_runtime：消息运行时依赖
catkin_package(CATKIN_DEPENDS
  roscpp rospy std_msgs message_runtime
  pcl_ros
)

#---------------------------------------------------------------------------------------
# Set default build to release
#---------------------------------------------------------------------------------------
# 如果未手动指定CMAKE_BUILD_TYPE（编译模式），则默认设置为Release模式
# CACHE STRING：将值存入CMake缓存，FORCE：强制覆盖原有缓存值
# Release模式开启编译器优化，Debug模式用于调试，这里默认选Release提升性能
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE "Release" CACHE STRING "Choose Release or Debug" FORCE)
endif()

#---------------------------------------------------------------------------------------
# Compiler config
#---------------------------------------------------------------------------------------
# 设置C++编译标准为C++14，确保代码兼容C++14特性
set(CMAKE_CXX_STANDARD 14)

# 强制要求编译器支持C++14标准，不支持则报错
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 禁用编译器特定的C++扩展，保证代码跨编译器可移植（如GNU扩展）
set(CMAKE_CXX_EXTENSIONS OFF)

## make sure the livox_lidar_sdk_static library is installed
# 查找Livox激光雷达SDK的静态库（liblivox_lidar_sdk_static.a），指定查找路径为/usr/local/lib
# 静态库会被编译进可执行文件，无需运行时依赖动态库
find_library(LIVOX_LIDAR_SDK_LIBRARY  liblivox_lidar_sdk_static.a    /usr/local/lib)

## PCL library
# 添加PCL库的链接目录，让编译器能找到PCL的库文件
link_directories(${PCL_LIBRARY_DIRS})

# 添加PCL的预编译宏定义，适配PCL的编译要求（如处理不同平台/版本的差异）
add_definitions(${PCL_DEFINITIONS})

#---------------------------------------------------------------------------------------
# generate excutable and add libraries
#---------------------------------------------------------------------------------------
# 创建可执行文件，名称为${PROJECT_NAME}_node（即livox_ros_driver2_node）
# 初始为空（""），后续通过target_sources添加源码文件，是CMake现代写法
add_executable(${PROJECT_NAME}_node
  ""
)

#---------------------------------------------------------------------------------------
# precompile macro and compile option
#---------------------------------------------------------------------------------------
# 为可执行文件添加编译选项：仅当编译器为GNU（gcc/g++）时，添加-Wall（显示所有警告）
# PRIVATE：该选项仅作用于当前目标，不传递给依赖该目标的其他目标
# $<$<CXX_COMPILER_ID:GNU>:-Wall>：CMake条件表达式，仅GNU编译器生效
target_compile_options(${PROJECT_NAME}_node
  PRIVATE $<$<CXX_COMPILER_ID:GNU>:-Wall>
)

#---------------------------------------------------------------------------------------
# add projects that depend on
#---------------------------------------------------------------------------------------
# 设置可执行文件的依赖：必须先编译${PROJECT_NAME}_generate_messages_cpp（消息生成目标）
# 确保自定义消息的C++代码生成完成后，再编译节点源码，避免头文件未生成的错误
add_dependencies(${PROJECT_NAME}_node ${PROJECT_NAME}_generate_messages_cpp)

#---------------------------------------------------------------------------------------
# source file
#---------------------------------------------------------------------------------------
# 为可执行文件添加源码文件，PRIVATE表示这些源码仅属于当前目标，不对外暴露
# 按功能模块划分：核心驱动（driver_node/lds等）、通信模块（comm）、配置解析（parse_cfg_file）、回调函数（call_back）
target_sources(${PROJECT_NAME}_node
  PRIVATE
  src/driver_node.cpp
  src/lds.cpp
  src/lds_lidar.cpp
  src/lddc.cpp
  src/livox_ros_driver2.cpp

  src/comm/comm.cpp
  src/comm/ldq.cpp
  src/comm/semaphore.cpp
  src/comm/lidar_imu_data_queue.cpp
  src/comm/cache_index.cpp
  src/comm/pub_handler.cpp

  src/parse_cfg_file/parse_cfg_file.cpp
  src/parse_cfg_file/parse_livox_lidar_cfg.cpp

  src/call_back/lidar_common_callback.cpp
  src/call_back/livox_lidar_callback.cpp
)

#---------------------------------------------------------------------------------------
# include file
#---------------------------------------------------------------------------------------
# 设置可执行文件的头文件包含目录，PUBLIC表示该目录会传递给依赖该目标的其他目标
# catkin_INCLUDE_DIRS：ROS1组件的头文件目录；PCL_INCLUDE_DIRS：PCL头文件目录
# APR_INCLUDE_DIRS：APR库头文件目录；3rdparty：第三方库头文件目录；src：源码根目录
target_include_directories(${PROJECT_NAME}_node
  PUBLIC
  ${catkin_INCLUDE_DIRS}
  ${PCL_INCLUDE_DIRS}
  ${APR_INCLUDE_DIRS}
  3rdparty
  src
)

#---------------------------------------------------------------------------------------
# link libraries
#---------------------------------------------------------------------------------------
# 为可执行文件链接依赖库：
# LIVOX_LIDAR_SDK_LIBRARY：Livox激光雷达SDK静态库；Boost_LIBRARY：Boost库
# catkin_LIBRARIES：ROS1组件库；PCL_LIBRARIES：PCL库；APR_LIBRARIES：APR库
target_link_libraries(${PROJECT_NAME}_node
  ${LIVOX_LIDAR_SDK_LIBRARY}
  ${Boost_LIBRARY}
  ${catkin_LIBRARIES}
  ${PCL_LIBRARIES}
  ${APR_LIBRARIES}
)


#---------------------------------------------------------------------------------------
# Install
#---------------------------------------------------------------------------------------

# 安装可执行文件到指定目录：
# ARCHIVE：静态库安装路径；LIBRARY：动态库安装路径；RUNTIME：可执行文件安装路径
# CATKIN_PACKAGE_LIB_DESTINATION：ROS1库文件默认路径；CATKIN_PACKAGE_BIN_DESTINATION：ROS1可执行文件默认路径
install(TARGETS ${PROJECT_NAME}_node
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
)

# 安装launch_ROS1目录下的所有文件/子目录到ROS1包共享目录的launch_ROS1子目录
# CATKIN_PACKAGE_SHARE_DESTINATION：ROS1包共享文件（如launch/cfg）的默认路径
install(DIRECTORY launch_ROS1/
  DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/launch_ROS1
)
```
