class external_cali {
public:
    // 构造函数：传入单个PnP匹配数据（激光点XYZ + 图像点UV）
    external_cali(PnPData p) {
        pd = p;  // 将匹配数据存储到类成员变量pd
    }

    // 重载()运算符：Ceres要求的代价函数计算接口（模板函数支持自动求导）
    template <typename T>
    bool operator()(const T *_q, const T *_t, T *residuals) const {
        // 1. 将相机内参矩阵转换为模板类型T（适配Ceres的自动求导精度）
        Eigen::Matrix<T, 3, 3> innerT = inner.cast<T>();
        
        // 2. 解析优化变量：四元数（_q[3]是w，_q[0-2]是x,y,z）+ 平移向量（_t[0-2]是tx,ty,tz）
        Eigen::Quaternion<T> q_incre{_q[3], _q[0], _q[1], _q[2]};  // 四元数存储顺序：w,x,y,z
        Eigen::Matrix<T, 3, 1> t_incre{_t[0], _t[1], _t[2]};       // 平移向量
        
        // 3. 激光点 → 相机坐标系转换（旋转+平移）
        Eigen::Matrix<T, 3, 1> p_l(T(pd.x), T(pd.y), T(pd.z));     // 激光点（世界/激光坐标系）
        Eigen::Matrix<T, 3, 1> p_c = q_incre.toRotationMatrix() * p_l + t_incre;  // 转换到相机坐标系
        
        // 4. 相机坐标系 → 图像平面投影（内参矩阵乘法，齐次坐标归一化）
        Eigen::Matrix<T, 3, 1> p_2 = innerT * p_c;                 // 投影后的齐次坐标（u*z, v*z, z）
        T u_proj = p_2[0] / p_2[2];  // 归一化得到投影像素u
        T v_proj = p_2[1] / p_2[2];  // 归一化得到投影像素v
        
        // 5. 计算残差（投影误差：预测UV - 实际UV）
        residuals[0] = u_proj - T(pd.u);
        residuals[1] = v_proj - T(pd.v);

        return true;  // 代价函数计算成功
    }

    // 静态函数：创建代价函数实例（Ceres要求的工厂模式，方便添加到优化问题）
    static ceres::CostFunction *Create(PnPData p) {
        // AutoDiffCostFunction：自动求导代价函数
        // 参数说明：<代价函数类, 残差维度(2:u,v), 优化变量1维度(4:四元数), 优化变量2维度(3:平移)>
        return (new ceres::AutoDiffCostFunction<external_cali, 2, 4, 3>(new external_cali(p)));
    }

private:
    PnPData pd;  // 类成员变量：存储单个PnP匹配数据（激光XYZ + 图像UV）
};



void getParameters() {
    cout << "Get the parameters from the launch file" << endl;

    // 从ROS参数服务器读取「激光数据路径」，读取失败则报错退出
    if (!ros::param::get("input_lidar_path", lidar_path)) {
        cout << "Can not get the value of input_lidar_path" << endl;
        exit(1);
    }
    // 读取「图像数据路径」
    if (!ros::param::get("input_photo_path", photo_path)) {
        cout << "Can not get the value of input_photo_path" << endl;
        exit(1);
    }
    // 读取「相机内参文件路径」
    if (!ros::param::get("intrinsic_path", intrinsic_path)) {
        cout << "Can not get the value of intrinsic_path" << endl;
        exit(1);
    }
    // 读取「外参输出文件路径」
    if (!ros::param::get("extrinsic_path", extrinsic_path)) {
        cout << "Can not get the value of extrinsic_path" << endl;
        exit(1);
    }
    // 读取「重投影误差阈值」
    if (!ros::param::get("error_threshold", error_threshold)) {
        cout << "Can not get the value of error_threshold" << endl;
        exit(1);
    }
    // 读取「外参初始化值」（12个元素：3x3旋转矩阵 + 3x1平移向量）
    init.resize(12);  // 初始化向量大小为12
    if (!ros::param::get("init_value", init)) {
        cout << "Can not get the value of init_value" << endl;
        exit(1);
    }
}


int main(int argc, char **argv) {
    // 1. 初始化ROS节点，节点名为「getExt1」
    ros::init(argc, argv, "getExt1");
    // 2. 读取配置参数（路径、阈值、初始化值）
    getParameters();

    // 3. 读取PnP匹配数据（激光点云与图像的对应点对）
    vector<PnPData> pData;  // 存储所有PnP匹配数据（每个元素是一个点对）
    getData(lidar_path, photo_path, pData);  // 自定义函数（common.h）：加载匹配数据

    // 4. 加载相机内参矩阵
    Eigen::Matrix4d extrin;  // 外参矩阵（4x4，此处仅作为临时变量）
    vector<float> intrinsic; // 存储内参数据（从文件读取的9个元素）
    getIntrinsic(intrinsic_path, intrinsic);  // 自定义函数（common.h）：读取内参文件
    // 将内参向量赋值给3x3内参矩阵inner（按行排列：fx, s, cx; 0, fy, cy; 0, 0, 1）
    inner << intrinsic[0], intrinsic[1], intrinsic[2],
             intrinsic[3], intrinsic[4], intrinsic[5],
             intrinsic[6], intrinsic[7], intrinsic[8];

    // 5. 初始化外参优化变量（旋转矩阵→四元数 + 平移向量）
    Eigen::Matrix3d R;  // 初始旋转矩阵（3x3）
    // 从init向量读取初始旋转矩阵（init[0-2]：第一行；init[4-6]：第二行；init[8-10]：第三行）
    R << init[0], init[1], init[2],
         init[4], init[5], init[6],
         init[8], init[9], init[10];
    Eigen::Quaterniond q(R);  // 将旋转矩阵转换为四元数（避免旋转矩阵正交约束）
    
    double ext[7];  // 优化变量数组（7个元素：四元数x,y,z,w + 平移tx,ty,tz）
    ext[0] = q.x();  // 四元数x
    ext[1] = q.y();  // 四元数y
    ext[2] = q.z();  // 四元数z
    ext[3] = q.w();  // 四元数w
    ext[4] = init[3];  // 平移tx（init[3]对应tx）
    ext[5] = init[7];  // 平移ty（init[7]对应ty）
    ext[6] = init[11]; // 平移tz（init[11]对应tz）

    // 将ext数组映射为Eigen四元数和平移向量（方便后续操作，无需拷贝数据）
    Eigen::Map<Eigen::Quaterniond> m_q = Eigen::Map<Eigen::Quaterniond>(ext);
    Eigen::Map<Eigen::Vector3d> m_t = Eigen::Map<Eigen::Vector3d>(ext + 4);

    // 6. 配置Ceres优化问题
    ceres::LocalParameterization *q_parameterization = new ceres::EigenQuaternionParameterization();  // 四元数参数化（保证单位四元数约束）
    ceres::Problem problem;  // 优化问题实例

    // 添加优化变量：四元数（4维，带单位约束）+ 平移向量（3维，无约束）
    problem.AddParameterBlock(ext, 4, q_parameterization);  // 四元数变量
    problem.AddParameterBlock(ext + 4, 3);                  // 平移向量变量（ext+4指向tx的地址）

    // 为每个PnP匹配点添加残差块（代价函数）
    for (auto val : pData) {
        ceres::CostFunction *cost_function = external_cali::Create(val);  // 创建代价函数
        // 添加残差块：代价函数 + 核函数（此处为NULL，无鲁棒核） + 关联的优化变量
        problem.AddResidualBlock(cost_function, NULL, ext, ext + 4);
    }

    // 7. 配置Ceres求解器选项
    ceres::Solver::Options options;
    options.linear_solver_type = ceres::DENSE_SCHUR;  // 线性求解器类型（稠密Schur补，适合小规模问题）
    options.minimizer_progress_to_stdout = true;      // 打印优化过程到控制台
    options.trust_region_strategy_type = ceres::LEVENBERG_MARQUARDT;  // 信任域策略（L-M算法，适合非线性最小二乘）

    // 8. 执行优化
    ceres::Solver::Summary summary;  // 优化结果摘要
    ceres::Solve(options, &problem, &summary);  // 求解优化问题
    cout << summary.BriefReport() << endl;  // 打印优化摘要（迭代次数、最终误差等）

    // 9. 保存并打印标定结果
    Eigen::Matrix3d rot = m_q.toRotationMatrix();  // 优化后的四元数→旋转矩阵
    writeExt(extrinsic_path, rot, m_t);  // 自定义函数（common.h）：保存外参（旋转矩阵+平移向量）到文件
    cout << "优化后的旋转矩阵：" << endl << rot << endl;
    cout << "优化后的平移向量：" << endl << m_t << endl;

    // 10. 验证标定结果（计算重投影误差）
    cout << "Use the extrinsic result to reproject the data" << endl;
    float error[2] = {0, 0};  // 存储u、v方向的平均重投影误差
    // 自定义函数（result_verify.h）：计算重投影误差
    getUVError(intrinsic_path, extrinsic_path, lidar_path, photo_path, error, error_threshold);
    
    // 打印误差结果
    cout << "u average error is: " << error[0] << endl;
    cout << "v average error is: " << error[1] << endl;
    // 判断误差是否小于阈值（合格则输出OK，否则提示不合格）
    if (error[0] + error[1] < error_threshold) {
        cout << endl << "The reprojection error smaller than the threshold, extrinsic result seems ok" << endl;
    } else {
        cout << endl << "The reprojection error bigger than the threshold, extrinsic result seems not ok" << endl;
    }

    return 0;  // 程序正常退出
}
